waystation_laser_battery = {
	icon = "GFX_arkship_laser_battery"
	section = "BATTERY_STARBASE_SECTION"
	construction_days = @waystation_build_time_180
	starbase_type = waystation
	category = sm_defense_cat

	potential = {
		is_waystation_starbase = yes
		exists = owner
		can_build_defensive_waystation_module = yes
		owner? = {
			has_technology = tech_lasers_1
		}
	}

	show_in_tech = "tech_lasers_1"

	show_tech_unlock_if = {
		is_nomadic = yes
	}

	station_modifier = {
		ship_hull_mult = @waystation_modifier_mult_10_percent
		ship_armor_mult = @waystation_modifier_mult_10_percent
	}

	triggered_station_modifier = {
		potential = {
			owner? = {
				has_country_flag = sacred_path_embrace_chaos
			}
		}
		custom_tooltip = embrace_chaos_waystations_tt
		show_only_custom_tooltip = no
		weapon_type_energy_weapon_damage_mult = @waystation_modifier_mult_5_percent
		weapon_type_energy_weapon_fire_rate_mult = @waystation_modifier_mult_5_percent
		ship_hull_mult = @waystation_modifier_mult_10_percent
		ship_armor_mult = @waystation_modifier_mult_10_percent
	}

	resources = {
		category = starbase_waystations_modules
		cost = {
			alloys = @waystation_cost_alloys_150
		}
		upkeep = {
			trade = @waystation_upkeep_trade_2
		}
	}

	triggered_component_set = {
		potential = {
			owner? = {
				country_uses_bio_ships = no
			}
		}
		component_set = RED_LASER
	}
	triggered_component_set = {
		potential = {
			owner? = {
				country_uses_bio_ships = yes
			}
		}
		component_set = RED_BIO_LASER
	}
	show_component_tooltips = yes
	custom_tooltip_with_modifiers = sm_waystation_laser_battery_tt

	ai_weight = {
		weight = 100 #use same weight here - ai weighting is done in starbase_types.txt
		modifier = {
			factor = 2
			ai_owner_prefers_energy_weapons = yes
		}
	}
}
