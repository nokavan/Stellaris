	}

			}
		}
	}

		is_nomadic = yes

		upkeep = {
			trade = @waystation_upkeep_trade_2
		}

		energy = {
			always = yes
		}
	}

	station_modifier = {
		custom_tooltip = allows_energy_collection
		show_only_custom_tooltip = no
		starbase_stockpile_energy_capacity_add = @waystation_stockpile_500
	}

	triggered_system_modifier = {
		potential = {
			exists = owner
			space_owner? = {
				NOT = {
					benefits_from_waystation_network = {
						NOMAD = root.owner
						SYSTEM_OWNER = this
					}
				}
			}
		}
		custom_tooltip = waystation_collection_penalty_energy_tt
		show_only_custom_tooltip = yes
		station_gatherers_energy_produces_mult = @waystation_modifier_negative_mult_25_percent
	}

	triggered_station_modifier = {
