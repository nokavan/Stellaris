# Синдикат Сопротивления × мод о Предтечах
# Империи с цивиками «Партизанская сеть» или «Теневой сюзерен» получают
#  - tech_forerunner_start_1: открывает ветку оружия Предтеч (лучи Стражей,
#    пушки, ракеты, торсионные и ионные орудия, истребители) и их корабельные
#    модули (щиты, броня из жёсткого света, реакторы, двигатели, сенсоры);
#  - tech_forerunner_sentinel_beam: первое оружие ветки (тир 2) — сразу, без исследования.
# Звуки и графика оружия берутся из самого мода о Предтечах.
#####################################################

namespace = ps_fr

country_event = {
	id = ps_fr.1
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		OR = {
			has_valid_civic = civic_ps_partisan_network
			has_valid_civic = civic_ps_shadow_overlord
		}
		NOT = { has_country_flag = ps_forerunner_arsenal }
	}

	immediate = {
		set_country_flag = ps_forerunner_arsenal
		give_technology = { message = no tech = tech_forerunner_start_1 }
		give_technology = { message = no tech = tech_forerunner_sentinel_beam }
		if = {
			limit = { is_ai = no }
			country_event = { id = ps_fr.2 days = 1 }
		}
	}
}

country_event = {
	id = ps_fr.2
	title = ps_fr.2.name
	desc = ps_fr.2.desc
	picture = GFX_ps_evt_spy
	is_triggered_only = yes

	option = {
		name = ps_fr.2.a
	}
}
