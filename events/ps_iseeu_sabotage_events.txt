espionage_operation_event = {
	id = iseeu_sabotage.1
	title = iseeu_sabotage.1.title
	desc = iseeu_sabotage.1.desc
	show_sound = station_explosion_small
	espionage_operation = yes
	picture = GFX_evt_large_explosion
	is_triggered_only = yes
	option = {
		name = iseeu_sabotage.1.fireworks
	}
	after = {
		if = {
			limit = {
				exists = target
				target = {
					solar_system = {
						NOT = {
							any_system_planet = {
								is_colony = yes
							}
						}
					}
				}
			}
			target.solar_system = {
				every_fleet_in_system = {
					limit = {
						OR = {
							is_ship_size = mining_station
							is_ship_size = research_station
							is_ship_class = shipclass_starbase
						}
					}
					destroy_fleet = this
				}
			}
			spynetwork = {
				add_spy_network_level = -20
			}
		}
		destroy_espionage_operation = this
	}
}

espionage_operation_event = {
	id = iseeu_sabotage.15
	title = iseeu_sabotage.15.title
	desc = iseeu_sabotage.15.desc
	show_sound = station_explosion_small
	espionage_operation = yes
	picture = GFX_evt_night_raid
	is_triggered_only = yes
	immediate = {
		target = {
			random_owned_planet = {
				limit = {
					has_planet_flag = iseeu_operation_bomb_planet_target_by_@root.owner
				}
				save_event_target_as = iseeu_target
				remove_planet_flag = iseeu_operation_bomb_planet_target_by_@root.owner
				# log = "old target name = [iseeu_target.GetName]"
			}
		}
	}
	option = {
		name = iseeu_sabotage.15.option
	}
	after = {
		if = {
			limit = {
				exists = event_target:iseeu_target
				event_target:iseeu_target = {
					exists = owner
				}
			}
			# log = "target exists!"
			# log = "target name = [iseeu_target.GetName]"
			event_target:iseeu_target = {
				if = {
					limit = {
						has_district = district_city
					}
					remove_district = district_city
				}
				else_if = {
					limit = {
						has_district = district_arcology_housing
					}
					remove_district = district_arcology_housing
				}
				else_if = {
					limit = {
						has_district = district_nexus
					}
					remove_district = district_nexus
				}
				else_if = {
					limit = {
						has_district = district_hive
					}
					remove_district = district_hive
				}
			}
			event_target:iseeu_target = {
				if = {
					limit = {
						any_owned_pop_group = { always = yes }
					}
					random_owned_pop_group = {
						kill_single_pop = yes
					}
				}
				add_deposit = d_iseeu_bomb_crater
				add_planet_devastation = 30
				hidden_effect = {
					owner = {
						country_event = {
							id = iseeu_sabotage.16
						}
					}
				}
			}
			spynetwork = {
				add_spy_network_level = -10
			}
		}
		destroy_espionage_operation = this
	}
}
