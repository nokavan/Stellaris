# Ежегодно: Буря может вспыхнуть в империи ИИ в поздней игре (раз за игру)
country_event = {
	id = ps_storm.40
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = yes
		is_country_type = default
		years_passed > 150
		num_owned_planets > 5
		NOT = { has_global_flag = ps_storm_happened }
		NOT = { has_global_flag = ps_storm_disabled }
	}

	immediate = {
		random_list = {
			1 = {
				modifier = { factor = 10 ps_is_partisan_empire = yes }
				modifier = { factor = 3 any_owned_planet = { planet_stability < 30 } }
				ps_storm_ignite = yes
			}
			400 = { }
		}
	}
}

# 1. Искра: смута в каждой империи
situation_event = {
	id = ps_storm.11
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		owner = {
			save_event_target_as = ps_storm
			save_event_target_as = ps_sponsor
		}
		set_global_flag = ps_storm_level_1
		every_playable_country = {
			limit = { ps_storm_target = yes }
			random_owned_planet = {
				limit = { is_capital = no }
				add_modifier = { modifier = ps_sown_dissent days = 1800 }
				carrier_event = { id = ps_revolt.50 }
			}
			country_event = { id = ps_storm.3 }
		}
		owner = { country_event = { id = ps_storm.3 } }
	}
}

# ЗАРАЗА ПОБЕДИЛА У ЖЕРТВЫ (scope = ситуация; owner = жертва)
#####################################################
situation_event = {
	id = ps_storm.21
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		random_playable_country = {
			limit = { has_country_flag = ps_red_storm }
			save_event_target_as = ps_storm
		}
		if = {
			limit = { exists = event_target:ps_storm }
			owner = {
				while = {
					count = 3
					if = {
						limit = {
							any_owned_planet = {
								is_capital = no
								NOT = { has_planet_flag = ps_storm_defecting }
							}
						}
						random_owned_planet = {
							limit = {
								is_capital = no
								NOT = { has_planet_flag = ps_storm_defecting }
							}
							set_planet_flag = ps_storm_defecting
							carrier_event = { id = ps_storm.60 }
						}
					}
					else = {
						# Осталась только столица — пала и она
						random_owned_planet = {
							limit = { NOT = { has_planet_flag = ps_storm_defecting } }
							set_planet_flag = ps_storm_defecting
							carrier_event = { id = ps_storm.60 }
						}
					}
				}
				country_event = { id = ps_storm.22 }
			}
		}
		# Зараза не уходит: отсчёт начинается заново
		add_situation_progress = -60
	}
}

# ЕЖЕГОДНО У БУРИ
#####################################################
country_event = {
	id = ps_storm.30
	hide_window = yes
	is_triggered_only = yes

	trigger = { has_country_flag = ps_red_storm }

	immediate = {
		save_event_target_as = ps_storm

		# С 3-го уровня зараза приходит в каждую империю без иммунитета
		if = {
			limit = { has_global_flag = ps_storm_contagion_active }
			every_playable_country = {
				limit = {
					ps_storm_target = yes
					NOT = { has_country_flag = ps_storm_immune }
					NOT = { any_situation = { is_situation_type = situation_ps_revolutionary_contagion } }
				}
				start_situation = { type = situation_ps_revolutionary_contagion }
			}
		}

		# 5-й уровень: каждая империя ежегодно теряет планету
		if = {
			limit = { has_country_flag = ps_storm_final }
			every_playable_country = {
				limit = { ps_storm_target = yes }
				if = {
					limit = { any_owned_planet = { is_capital = no } }
					random_owned_planet = {
						limit = { is_capital = no }
						set_planet_flag = ps_storm_defecting
						carrier_event = { id = ps_storm.60 }
					}
				}
				else = {
					random_owned_planet = {
						set_planet_flag = ps_storm_defecting
						carrier_event = { id = ps_storm.60 }
					}
				}
			}
		}
	}
}

#####################################################
# ПЛАНЕТА ПЕРЕХОДИТ К БУРЕ (scope = планета)
#####################################################
carrier_event = {
	id = ps_storm.60
	hide_window = yes
	is_triggered_only = yes

	trigger = { exists = event_target:ps_storm }

	immediate = {
		remove_planet_flag = ps_storm_defecting
		set_owner = event_target:ps_storm
		set_controller = event_target:ps_storm
		add_modifier = { modifier = ps_storm_liberated_world days = 1800 }
		event_target:ps_storm = {
			set_timed_country_flag = { flag = ps_storm_recent_victory days = 360 }
		}
	}
}
