#####################################################
# Синдикат Сопротивления — КРИЗИС «КРАСНАЯ БУРЯ»
# Восстание свергает своего хозяина, затем по одному объявляет войну соседям.
# Партизанские империи игроков и их вассалов Буря не трогает.
#
#  ps_storm.1   операция «Красная Буря»: финальный этап (scope = операция)
#  ps_storm.2   всем: Буря родилась
#  ps_storm.3   всем: Буря выросла на уровень (scope = ситуация)
#  ps_storm.5   РОЖДЕНИЕ БУРИ (scope = планета хозяина)
#  ps_storm.6   жертве: Буря объявила войну
#  ps_storm.7   всем: Буря выбрала новую цель
#  ps_storm.8   всем: Буря победила в войне
#  ps_storm.30  ежегодно у Бури: следующая цель
#  ps_storm.40  ежегодно: Буря может вспыхнуть сама в империи ИИ
#  ps_storm.50  Буря уничтожена (scope = ситуация)
#  ps_storm.51  всем: Буря утихла
#####################################################

namespace = ps_storm

#####################################################
# ЗАПУСК
#####################################################

# Финал операции «Красная Буря» (scope = операция; owner = партизаны; target = хозяин)
espionage_operation_event = {
	id = ps_storm.1
	title = ps_storm.1.name
	desc = ps_storm.1.desc
	espionage_operation = yes
	picture = GFX_ps_evt_partisans
	show_sound = event_espionage_concluded
	is_triggered_only = yes

	immediate = {
		owner = { save_event_target_as = ps_storm_sponsor }
		target = {
			if = {
				limit = { any_owned_planet = { is_capital = no planet_stability < 40 } }
				random_owned_planet = {
					limit = { is_capital = no planet_stability < 40 }
					save_event_target_as = ps_storm_cradle
				}
			}
			else = {
				random_owned_planet = {
					limit = { is_capital = no }
					save_event_target_as = ps_storm_cradle
				}
			}
		}
	}

	option = {
		name = ps_storm.1.a
		custom_tooltip = ps_storm.1.a.tooltip
		if = {
			limit = { exists = event_target:ps_storm_cradle }
			event_target:ps_storm_cradle = { carrier_event = { id = ps_storm.5 } }
		}
	}

	after = { destroy_espionage_operation = this }
}

# Ежегодно: Буря может вспыхнуть сама в крупной империи ИИ (раз за игру)
country_event = {
	id = ps_storm.40
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_ai = yes
		is_country_type = default
		years_passed > 150
		num_owned_planets > 8
		ps_storm_target = yes
		NOT = { has_global_flag = ps_storm_happened }
		NOT = { has_global_flag = ps_storm_disabled }
	}

	immediate = {
		random_list = {
			1 = {
				modifier = { factor = 3 any_owned_planet = { planet_stability < 30 } }
				random_owned_planet = {
					limit = { is_capital = no }
					carrier_event = { id = ps_storm.5 }
				}
			}
			400 = { }
		}
	}
}

#####################################################
# РОЖДЕНИЕ БУРИ (scope = планета-колыбель; owner = хозяин)
#####################################################
carrier_event = {
	id = ps_storm.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		exists = owner
		NOT = { has_global_flag = ps_storm_happened }
	}

	immediate = {
		set_global_flag = ps_storm_happened
		owner = { save_event_target_as = ps_storm_host }

		# 1. Государство восставших
		create_country = {
			name = "NAME_ps_red_storm"
			type = default
			species = root.owner_species
			authority = random
			civics = random
			ethos = random
			ignore_initial_colony_error = yes
			effect = {
				save_event_target_as = ps_storm
				set_country_flag = ps_red_storm
			}
		}

		# 2. Колыбель и ещё до 4 неспокойных планет хозяина переходят к Буре
		set_owner = event_target:ps_storm
		set_controller = event_target:ps_storm
		add_modifier = { modifier = ps_storm_liberated_world days = 1800 }
		event_target:ps_storm_host = {
			while = {
				count = 4
				if = {
					limit = {
						any_owned_planet = {
							is_capital = no
							planet_stability < 50
						}
					}
					random_owned_planet = {
						limit = {
							is_capital = no
							planet_stability < 50
						}
						set_owner = event_target:ps_storm
						set_controller = event_target:ps_storm
						add_modifier = { modifier = ps_storm_liberated_world days = 1800 }
					}
				}
			}
			# 3. Мятеж: четверть флота хозяина переходит к восставшим
			create_fleet_from_naval_cap = 0.25
			last_created_fleet = {
				set_owner = event_target:ps_storm
			}
			set_country_flag = ps_storm_designated
		}

		# 4. Армии, ресурсы и рост Бури
		event_target:ps_storm = {
			every_owned_planet = {
				while = {
					count = 3
					create_army = {
						name = "NAME_ps_partisan_detachment"
						owner = event_target:ps_storm
						type = ps_partisan_insurgent
					}
				}
			}
			add_resource = { energy = 5000 minerals = 5000 alloys = 3000 food = 2000 influence = 300 }
			start_situation = { type = situation_ps_red_storm }
			if = {
				limit = { exists = event_target:ps_storm_sponsor }
				add_opinion_modifier = { who = event_target:ps_storm_sponsor modifier = opinion_ps_liberators }
			}
			# 5. Первая война — против хозяина
			declare_war = {
				target = event_target:ps_storm_host
				attacker_war_goal = wg_ps_red_storm
			}
		}

		# 6. Галактика узнаёт о Буре; все её будущие цели видят в ней смертельную угрозу
		every_playable_country = {
			limit = { ps_storm_target = yes }
			add_opinion_modifier = { who = event_target:ps_storm modifier = opinion_ps_red_storm }
		}
		every_playable_country = {
			limit = { is_country_type = default }
			country_event = { id = ps_storm.2 }
		}
	}
}

country_event = {
	id = ps_storm.2
	title = ps_storm.2.name
	desc = {
		trigger = { has_country_flag = ps_storm_designated }
		text = ps_storm.2.desc.host
	}
	desc = {
		trigger = { NOT = { has_country_flag = ps_storm_designated } }
		text = ps_storm.2.desc
	}
	picture = GFX_ps_evt_partisans
	is_triggered_only = yes

	option = { name = ps_storm.2.a }
}

#####################################################
# ПОХОД НА СОСЕДЕЙ — ежегодно у Бури
#####################################################
country_event = {
	id = ps_storm.30
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_country_flag = ps_red_storm
		is_at_war = no
		NOT = { has_country_flag = ps_storm_war_cooldown }
	}

	immediate = {
		save_event_target_as = ps_storm
		# Снять старую метку цели
		every_playable_country = {
			limit = { has_country_flag = ps_storm_designated }
			remove_country_flag = ps_storm_designated
		}
		# Следующая цель: сначала сосед, иначе любая знакомая империя
		if = {
			limit = { any_neighbor_country = { ps_storm_target = yes } }
			random_neighbor_country = {
				limit = { ps_storm_target = yes }
				save_event_target_as = ps_storm_next
			}
		}
		else = {
			random_playable_country = {
				limit = {
					ps_storm_target = yes
					has_communications = root
				}
				save_event_target_as = ps_storm_next
			}
		}
		if = {
			limit = { exists = event_target:ps_storm_next }
			event_target:ps_storm_next = {
				set_country_flag = ps_storm_designated
				country_event = { id = ps_storm.6 }
			}
			declare_war = {
				target = event_target:ps_storm_next
				attacker_war_goal = wg_ps_red_storm
			}
			every_playable_country = {
				limit = {
					is_country_type = default
					NOT = { has_country_flag = ps_storm_designated }
				}
				country_event = { id = ps_storm.7 }
			}
		}
	}
}
