# Мегаструктура построена или улучшена.
# root = страна, from = мегаструктура, fromfrom = система.
# Миры мегаструктуры могут появиться в on_build_complete чуть позже,
# поэтому только помечаем систему и проверяем её через 5 дней.
country_event = {
	id = ps_mega.5
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		ps_is_partisan_empire = yes
		ps_has_mega_doctrine = yes
		exists = fromfrom
	}

	immediate = {
		fromfrom = { set_star_flag = ps_mega_pending }
		country_event = { id = ps_mega.6 days = 5 }
	}
}

country_event = {
	id = ps_mega.6
	hide_window = yes
	is_triggered_only = yes

	trigger = { ps_is_partisan_empire = yes }

	immediate = {
		# Не только в границах: кочевники (без границ) строят мегаструктуры у своих стоянок
		every_system = {
			limit = { has_star_flag = ps_mega_pending }
			remove_star_flag = ps_mega_pending
			if = {
				limit = {
					root = { ps_has_mega_doctrine = yes }
					any_system_planet = {
						is_colonizable = yes
						NOT = { exists = owner }
					}
				}
				ps_mega_seed_system = yes
				root = { country_event = { id = ps_mega.7 } }
			}
		}
	}
}

# ТЕНЕВАЯ ЛОГИСТИКА: строители, инженерные суда и корабли снабжения
# партизан сами включают маскировку, если у них есть сила маскировки.
# Ручное отключение кнопкой из модов Cloakable Construction / Logistics Ships
# уважается (их флаги перезарядки).
country_event = {
	id = ps_mega.20
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		ps_is_partisan_empire = yes
		has_country_flag = ps_shadow_logistics
	}

	immediate = {
		every_owned_fleet = {
			limit = {
				ps_is_logistics_fleet = yes
				is_cloaked = no
				has_cloaking_strength > 0
				NOT = { has_fleet_flag = construction_cloak_cooldown }
				NOT = { has_fleet_flag = logistics_cloak_cooldown }
				NOT = { has_fleet_flag = ps_cloak_hold }
			}
			set_cloaking_active = yes
		}
	}
}
