# Ежегодно: пересчёт бонусов сюзерена, дань специализаций,
# предложение специализировать одного нового вассала.
# Вассалы перебираются через every_playable_country + overlord,
# как в ванильных событиях 4.x.
country_event = {
	id = ps.11
	hide_window = yes
	is_triggered_only = yes

	trigger = { ps_is_shadow_overlord = yes }

	immediate = {
		# Ванильные специалисты (Overlord) автоматически вписываются в сеть
		every_playable_country = {
			limit = {
				is_subject = yes
				overlord = { is_same_value = root }
				NOT = { has_country_flag = ps_specialized }
			}
			if = {
				limit = { is_specialist_subject_type = { TYPE = bulwark } }
				set_country_flag = ps_specialized
				set_country_flag = ps_spec_bastion
			}
			else_if = {
				limit = { is_specialist_subject_type = { TYPE = scholarium } }
				set_country_flag = ps_specialized
				set_country_flag = ps_spec_underground_lab
			}
			else_if = {
				limit = { is_specialist_subject_type = { TYPE = prospectorium } }
				set_country_flag = ps_specialized
				set_country_flag = ps_spec_black_market
			}
		}

		remove_modifier = ps_overlord_spy_ring
		remove_modifier = ps_overlord_bastion
		remove_modifier = ps_overlord_black_market
		remove_modifier = ps_overlord_underground_lab

		if = {
			limit = { any_playable_country = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_spy_ring } }
			add_modifier = { modifier = ps_overlord_spy_ring days = -1 }
		}
		if = {
			limit = { any_playable_country = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_bastion } }
			add_modifier = { modifier = ps_overlord_bastion days = -1 }
		}
		if = {
			limit = { any_playable_country = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_black_market } }
			add_modifier = { modifier = ps_overlord_black_market days = -1 }
		}
		if = {
			limit = { any_playable_country = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_underground_lab } }
			add_modifier = { modifier = ps_overlord_underground_lab days = -1 }
		}

		# Специализированная дань (за каждого вассала)
		every_playable_country = {
			limit = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_spy_ring }
			root = { add_resource = { influence = 15 } }
		}
		every_playable_country = {
			limit = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_bastion }
			root = { add_resource = { alloys = 75 } }
		}
		every_playable_country = {
			limit = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_black_market }
			root = { add_resource = { energy = 150 } }
		}
		every_playable_country = {
			limit = { is_subject = yes overlord = { is_same_value = root } has_country_flag = ps_spec_underground_lab }
			root = { add_resource = { physics_research = 50 society_research = 50 engineering_research = 50 } }
		}

		# Подстраховка: предложить роль одному вассалу без неё
		if = {
			limit = {
				any_playable_country = {
					is_subject = yes
					overlord = { is_same_value = root }
					NOT = { has_country_flag = ps_specialized }
				}
			}
			random_playable_country = {
				limit = {
					is_subject = yes
					overlord = { is_same_value = root }
					NOT = { has_country_flag = ps_specialized }
				}
				save_event_target_as = ps_subject
			}
			country_event = { id = ps.10 }
		}
	}
}

# Работают через ванильные шпионские сети: цель выбирается

country_event = {
	id = ps.20
	title = ps.20.name
	desc = ps.20.desc
	picture = GFX_ps_evt_spy
	show_sound = event_encrypted_comms
	is_triggered_only = yes

	trigger = {
		ps_is_partisan_empire = yes
		has_active_spynetwork = yes
		any_spynetwork = {
			owner = { is_same_value = root }
			is_spynetwork_level > 20
			target = { is_country_type = default }
		}
	}

	immediate = {
		random_spynetwork = {
			limit = {
				owner = { is_same_value = root }
				is_spynetwork_level > 20
				target = { is_country_type = default }
			}
			target = { save_event_target_as = ps_op_target }
		}
	}

	# Украсть чертежи
	option = {
		name = ps.20.a
		custom_tooltip = ps.20.cost_10
		hidden_effect = { ps_spend_network = { AMOUNT = -10 } }
		random_list = {
			70 = {
				modifier = { factor = 1.5 ps_has_spy_ring_subject = yes }
				add_monthly_resource_mult = {
					resource = physics_research
					value = 6
					min = 150
					max = 3000
				}
				add_monthly_resource_mult = {
					resource = engineering_research
					value = 6
					min = 150
					max = 3000
				}
			}
			30 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = { factor = 2 }
	}

	# Диверсия на верфях
	option = {
		name = ps.20.b
		custom_tooltip = ps.20.cost_15
		hidden_effect = { ps_spend_network = { AMOUNT = -15 } }
		random_list = {
			65 = {
				modifier = { factor = 1.5 ps_has_spy_ring_subject = yes }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_sabotaged_shipyards days = 1800 }
				}
			}
			35 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 3 is_at_war = yes }
		}
	}

	# Поднять подполье
	option = {
		name = ps.20.c
		custom_tooltip = ps.20.cost_20
		hidden_effect = { ps_spend_network = { AMOUNT = -20 } }
		random_list = {
			60 = {
				modifier = { factor = 1.5 ps_has_spy_ring_subject = yes }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_incited_unrest days = 1800 }
				}
				add_resource = { unity = 200 }
			}
			40 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = { factor = 1 }
	}

	# Обчистить казну
	option = {
		name = ps.20.d
		custom_tooltip = ps.20.cost_10
		hidden_effect = { ps_spend_network = { AMOUNT = -10 } }
		random_list = {
			60 = {
				modifier = {
					factor = 1.5
					any_playable_country = {
						is_subject = yes
						overlord = { is_same_value = root }
						has_country_flag = ps_spec_black_market
					}
				}
				add_resource = { energy = 400 }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_stolen_ledgers days = 1080 }
				}
			}
			40 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = { factor = 1 }
	}

	# Залечь на дно: сеть растёт
	option = {
		name = ps.20.e
		custom_tooltip = ps.20.e.tooltip
		hidden_effect = { ps_spend_network = { AMOUNT = 5 } }
		ai_chance = { factor = 0.5 }
	}
}

	id = ps.30
	hide_window = yes
	is_triggered_only = yes

		from = { ps_is_partisan_empire = yes }

	}

	immediate = {
		set_planet_flag = ps_resisting
		save_event_target_as = ps_occupied_planet

		fromfrom = {
			if = {
				limit = { NOT = { has_modifier = ps_partisan_harassment } }
				add_modifier = { modifier = ps_partisan_harassment days = 360 }
			}
		}

		from = {
			add_resource = { unity = 150 influence = 10 }
			if = {
				limit = { NOT = { has_country_flag = ps_resistance_notified } }
				set_timed_country_flag = { flag = ps_resistance_notified days = 720 }
				country_event = { id = ps.31 }
			}
		}
	}
}
