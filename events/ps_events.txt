# ВАССАЛЫ
# Сами специализации — настоящие типы вассалов (common/specialist_subject_*),
# опыт копится от лояльности, перки открываются на уровнях I–III.
# Перки ставят на вассала флаги ps_raiders_1..3 / ps_residency_active /
# ps_exchange_active, которые читают события ниже.
#####################################################

# Ежегодно у сюзерена: Присяга Синдиката и каперские рейды
country_event = {
	id = ps.14
	hide_window = yes
	is_triggered_only = yes

	trigger = { ps_is_shadow_overlord = yes }

	immediate = {
		# Присяга: +1 лояльность в месяц всем вассалам -> быстрее растёт опыт специализации
		if = {
			limit = { has_edict = edict_ps_oath }
			every_playable_country = {
				limit = {
					is_subject = yes
					overlord = { is_same_value = root }
				}
				add_modifier = { modifier = ps_oath_loyalty days = 370 }
			}
		}

		# Каперские рейды во время войны
		if = {
			limit = {
				is_at_war = yes
				any_playable_country = {
					is_subject = yes
					overlord = { is_same_value = root }
					OR = {
						has_country_flag = ps_raiders_1
						has_country_flag = ps_raiders_2
						has_country_flag = ps_raiders_3
					}
				}
			}
			every_playable_country = {
				limit = {
					is_subject = yes
					overlord = { is_same_value = root }
				}
				if = {
					limit = { has_country_flag = ps_raiders_3 }
					root = { add_resource = { energy = 300 } }
				}
				else_if = {
					limit = { has_country_flag = ps_raiders_2 }
					root = { add_resource = { energy = 200 } }
				}
				else_if = {
					limit = { has_country_flag = ps_raiders_1 }
					root = { add_resource = { energy = 100 } }
				}
			}
			every_playable_country = {
				limit = {
					is_at_war_with = root
					NOT = { has_modifier = ps_privateer_raids }
				}
				add_modifier = { modifier = ps_privateer_raids days = 360 }
			}
			if = {
				limit = { NOT = { has_country_flag = ps_raids_notified } }
				set_timed_country_flag = { flag = ps_raids_notified days = 1800 }
				country_event = { id = ps.15 }
			}
		}
	}
}

# Работают через ванильные шпионские сети: цель выбирается

country_event = {
	id = ps.20
	title = ps.20.name
	desc = ps.20.desc
	picture = GFX_ps_evt_spy
	show_sound = event_encrypted_comms
	is_triggered_only = yes

	trigger = {
		ps_is_partisan_empire = yes
		has_active_spynetwork = yes
		any_spynetwork = {
			owner = { is_same_value = root }
			is_spynetwork_level > 20
			target = { is_country_type = default }
		}
	}

	immediate = {
		random_spynetwork = {
			limit = {
				owner = { is_same_value = root }
				is_spynetwork_level > 20
				target = { is_country_type = default }
			}
			target = { save_event_target_as = ps_op_target }
		}
	}

	# Украсть чертежи
	option = {
		name = ps.20.a
		custom_tooltip = ps.20.cost_10
		hidden_effect = { ps_spend_network = { AMOUNT = -10 } }
		random_list = {
			70 = {
				modifier = { factor = 1.5 ps_has_spy_ring_subject = yes }
				add_monthly_resource_mult = {
					resource = physics_research
					value = 6
					min = 150
					max = 3000
				}
				add_monthly_resource_mult = {
					resource = engineering_research
					value = 6
					min = 150
					max = 3000
				}
			}
			30 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = { factor = 2 }
	}

	# Диверсия на верфях
	option = {
		name = ps.20.b
		custom_tooltip = ps.20.cost_15
		hidden_effect = { ps_spend_network = { AMOUNT = -15 } }
		random_list = {
			65 = {
				modifier = { factor = 1.5 ps_has_spy_ring_subject = yes }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_sabotaged_shipyards days = 1800 }
				}
			}
			35 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 3 is_at_war = yes }
		}
	}

	# Поднять подполье
	option = {
		name = ps.20.c
		custom_tooltip = ps.20.cost_20
		hidden_effect = { ps_spend_network = { AMOUNT = -20 } }
		random_list = {
			60 = {
				modifier = { factor = 1.5 ps_has_spy_ring_subject = yes }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_incited_unrest days = 1800 }
				}
				add_resource = { unity = 200 }
			}
			40 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = { factor = 1 }
	}

	# Обчистить казну
	option = {
		name = ps.20.d
		custom_tooltip = ps.20.cost_10
		hidden_effect = { ps_spend_network = { AMOUNT = -10 } }
		random_list = {
			60 = {

# КОВЧЕГ: СТОЯНКА СОПРОТИВЛЕНИЯ
# Пока ковчег стоит лагерем (is_system_locked), он работает
# с чужими колониями в своей системе:
#   союзникам  — помощь (ps_ark_aid),
#   остальным  — контрабанда и подполье (ps_ark_contraband).
# Модификаторы на 40 дней обновляются ежемесячно и сами
# спадают, когда ковчег уходит.
#####################################################

country_event = {
	id = ps.70
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		has_origin = origin_ps_partisan_ark
		any_owned_fleet = {
			is_arkship_fleet = yes
			exists = solar_system
			any_owned_ship = {
				is_arkship_ship = yes
				is_system_locked = yes
			}
		}
	}

	immediate = {
		every_owned_fleet = {
			limit = {
				is_arkship_fleet = yes
				exists = solar_system
				any_owned_ship = {
					is_arkship_ship = yes
					is_system_locked = yes
				}
			}
			solar_system = {
				every_system_planet_colony = {
					limit = {
						exists = owner
						owner = {
							is_country_type = default
							NOT = { is_same_value = root }
						}
					}
					if = {
						limit = {
							owner = {
								OR = {
									is_in_federation_with = root
									has_defensive_pact = root
									is_friendly_to = root
									AND = {
										is_subject = yes
										overlord = { is_same_value = root }
									}
								}
							}
						}
						add_modifier = { modifier = ps_ark_aid days = 40 }
						root = { add_resource = { unity = 15 influence = 1 } }
					}
					else = {
						add_modifier = { modifier = ps_ark_contraband days = 40 }
						root = { add_resource = { energy = 30 unity = 5 } }
					}
					if = {
						limit = { NOT = { root = { has_country_flag = ps_ark_camp_notified } } }
						save_event_target_as = ps_ark_planet
						root = {
							set_country_flag = ps_ark_camp_notified
							country_event = { id = ps.71 }
						}
					}
				}
			}
		}
	}
}
