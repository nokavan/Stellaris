# Ежегодно: пересчёт бонусов сюзерена, дань специализаций,
# предложение специализировать одного нового вассала.
country_event = {
	id = ps.11
	hide_window = yes
	is_triggered_only = yes

	trigger = { ps_is_shadow_overlord = yes }

	immediate = {
		remove_modifier = ps_overlord_spy_ring
		remove_modifier = ps_overlord_bastion
		remove_modifier = ps_overlord_black_market
		remove_modifier = ps_overlord_underground_lab

		if = {
			limit = { any_subject = { has_country_flag = ps_spec_spy_ring } }
			add_modifier = { modifier = ps_overlord_spy_ring days = -1 }
		}
		if = {
			limit = { any_subject = { has_country_flag = ps_spec_bastion } }
			add_modifier = { modifier = ps_overlord_bastion days = -1 }
		}
		if = {
			limit = { any_subject = { has_country_flag = ps_spec_black_market } }
			add_modifier = { modifier = ps_overlord_black_market days = -1 }
		}
		if = {
			limit = { any_subject = { has_country_flag = ps_spec_underground_lab } }
			add_modifier = { modifier = ps_overlord_underground_lab days = -1 }
		}

		# Специализированная дань (за каждого вассала)
		every_subject = {
			limit = { has_country_flag = ps_spec_spy_ring }
			root = { add_resource = { influence = 15 } }
		}
		every_subject = {
			limit = { has_country_flag = ps_spec_bastion }
			root = { add_resource = { alloys = 75 } }
		}
		every_subject = {
			limit = { has_country_flag = ps_spec_black_market }
			root = { add_resource = { energy = 150 } }
		}
		every_subject = {
			limit = { has_country_flag = ps_spec_underground_lab }
			root = { add_resource = { physics_research = 50 society_research = 50 engineering_research = 50 } }
		}

		# Предложить специализацию одному новому вассалу
		if = {
			limit = { any_subject = { NOT = { has_country_flag = ps_specialized } } }
			random_subject = {
				limit = { NOT = { has_country_flag = ps_specialized } }
				save_event_target_as = ps_subject
			}
			country_event = { id = ps.10 }
		}
	}
}

country_event = {
	id = ps.20
	title = ps.20.name
	desc = ps.20.desc
	picture = GFX_ps_evt_spy
	is_triggered_only = yes

	trigger = { ps_is_partisan_empire = yes }

	immediate = {
		random_country = {
			limit = {
				is_country_type = default
				NOT = { is_same_value = root }
				has_communications = root
				NOT = { is_subject = yes }
			}
			save_event_target_as = ps_op_target
		}
	}

	# Украсть технологические чертежи
	option = {
		name = ps.20.a
		trigger = {
			exists = event_target:ps_op_target
			resource_stockpile_compare = { resource = influence value >= 50 }
		}
		add_resource = { influence = -50 }
		random_list = {
			70 = {
				modifier = { factor = 1.5 any_subject = { has_country_flag = ps_spec_spy_ring } }
				add_monthly_resource_mult = {
					resource = physics_research
					value = 6
					min = 150
					max = 3000
				}
				add_monthly_resource_mult = {
					resource = engineering_research
					value = 6
					min = 150
					max = 3000
				}
			}
			30 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = { factor = 2 }
	}

	# Саботировать верфи
	option = {
		name = ps.20.b
		trigger = {
			exists = event_target:ps_op_target
			resource_stockpile_compare = { resource = energy value >= 300 }
		}
		add_resource = { energy = -300 }
		random_list = {
			65 = {
				modifier = { factor = 1.5 any_subject = { has_country_flag = ps_spec_spy_ring } }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_sabotaged_shipyards days = 1800 }
				}
			}

				country_event = { id = ps.21 }
			}
		}
		ai_chance = {
			factor = 1
			modifier = { factor = 3 is_at_war = yes }
		}
	}

	# Поднять подполье
	option = {
		name = ps.20.c
		trigger = {
			exists = event_target:ps_op_target
			resource_stockpile_compare = { resource = influence value >= 75 }
		}
		add_resource = { influence = -75 }
		random_list = {
			60 = {
				modifier = { factor = 1.5 any_subject = { has_country_flag = ps_spec_spy_ring } }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_incited_unrest days = 1800 }
				}
				add_resource = { unity = 200 }
			}
			40 = {
				country_event = { id = ps.21 }
			}
		}
		ai_chance = { factor = 1 }
	}

	# Обчистить казну
	option = {
		name = ps.20.d
		trigger = { exists = event_target:ps_op_target }
		random_list = {
			60 = {
				modifier = { factor = 1.5 any_subject = { has_country_flag = ps_spec_black_market } }
				add_resource = { energy = 400 }
				event_target:ps_op_target = {
					add_modifier = { modifier = ps_stolen_ledgers days = 1080 }
				}

			40 = {

			}
		}
		ai_chance = { factor = 1 }
	}

	# Залечь на дно
	option = {
		name = ps.20.e

# Ежемесячно: наши оккупированные планеты поднимают партизан
country_event = {
	id = ps.30
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		ps_is_partisan_empire = yes
		any_owned_planet = {
			OR = {
				AND = {
					NOT = { has_planet_flag = ps_resisting }
					exists = controller
					controller = { NOT = { is_same_value = root } }
				}
				AND = {
					has_planet_flag = ps_resisting
					controller = { is_same_value = root }
				}
			}
		}
	}

	immediate = {
		# Освобождённые планеты
		every_owned_planet = {
			limit = {
				has_planet_flag = ps_resisting
				controller = { is_same_value = root }
			}
			remove_planet_flag = ps_resisting
		}

		# Новые оккупированные планеты
		every_owned_planet = {
			limit = {
				NOT = { has_planet_flag = ps_resisting }
				exists = controller
				controller = { NOT = { is_same_value = root } }
			}
			set_planet_flag = ps_resisting
			controller = {
				if = {
					limit = { NOT = { has_modifier = ps_partisan_harassment } }
					add_modifier = { modifier = ps_partisan_harassment days = 360 }
				}
			}
			root = {
				add_resource = { unity = 150 influence = 10 }
			}
			if = {
				limit = { NOT = { root = { has_country_flag = ps_resistance_notified } } }
				save_event_target_as = ps_occupied_planet
				root = {
					set_timed_country_flag = { flag = ps_resistance_notified days = 720 }
					country_event = { id = ps.31 }
				}
			}
		}
	}
}
