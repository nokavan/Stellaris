	espionage_operation = yes
	picture = GFX_evt_spy_network
	show_sound = event_espionage_concluded
	is_triggered_only = yes

	immediate = {
		target = {
			random_owned_leader = {
				save_event_target_as = ps_liquidation_target
			}
		}
	}

	option = {
		name = ps_sabotage.2.a
		custom_tooltip = ps_sabotage.2.a.tooltip

			limit = { exists = event_target:ps_liquidation_target }
			event_target:ps_liquidation_target = {
				kill_leader = { show_notification = yes }
			}
		}
	}

	after = {
		destroy_espionage_operation = this
	}
}

espionage_operation_event = {
	id = ps_sabotage.4
	title = ps_sabotage.4.name
	desc = ps_sabotage.4.desc
	espionage_operation = yes
	picture = GFX_evt_alien_propaganda
	show_sound = event_espionage_concluded
	is_triggered_only = yes

	immediate = {
		owner = { save_event_target_as = ps_sponsor }
		target = {
			random_owned_planet = {
				limit = {
					is_capital = no
					planet_stability < 50
				}
				save_event_target_as = ps_revolt_planet
			}
		}
	}

	option = {
		name = ps_sabotage.4.a
		custom_tooltip = ps_sabotage.4.a.tooltip
		if = {
			limit = { exists = event_target:ps_revolt_planet }
			event_target:ps_revolt_planet = {
				add_modifier = { modifier = ps_armed_underground days = 1080 }
				# Пометить как поддержанную Синдикатом (см. ps_revolt.50)
				carrier_event = { id = ps_revolt.50 }
				# Высадить армии повстанцев
				carrier_event = { id = ps_sabotage.50 }
			}
		}
	}

	after = {
		destroy_espionage_operation = this
	}
}

# Планета: создать скрытую страну повстанцев и её армии (идея Skullduggery «Riot»)
carrier_event = {
	id = ps_sabotage.50
	hide_window = yes
	is_triggered_only = yes

	immediate = {
		create_country = {
			name = "NAME_ps_partisan_insurgents"
			species = root.owner_species
			type = "ps_partisan_insurgents"
			ignore_initial_colony_error = yes
			name_list = "Prethoryn"
			flag = {
				background = {
					category = "backgrounds"
					file = "new_dawn.dds"
				}
				colors = {
					"red"
					"black"
					"null"
					"null"
				}
			}
			effect = {
				save_event_target_as = ps_insurgents
			}
		}
		# Базово 3 отряда; больше, если на планете уже есть смута и подполье
		while = {
			count = 3
			create_army = {
				name = "NAME_ps_partisan_detachment"
				owner = event_target:ps_insurgents
				type = ps_partisan_insurgent
			}
		}
		if = {
			limit = { has_modifier = ps_sown_dissent }
			while = {
				count = 2
				create_army = {
					name = "NAME_ps_partisan_detachment"
					owner = event_target:ps_insurgents
					type = ps_partisan_insurgent
				}
			}
		}
		if = {
			limit = { has_planet_flag = ps_partisan_sponsored }
			create_army = {
				name = "NAME_ps_partisan_detachment"
				owner = event_target:ps_insurgents
				type = ps_partisan_insurgent
			}
		}
		owner = { country_event = { id = ps_sabotage.20 } }
	}
}

						alloys = -10

					}

			}
			country_event = { id = ps_sabotage.21 }
		}
	}

	after = {
