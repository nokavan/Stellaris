# СМЕНА ПРИСЯГИ: чужой вассал теряет лояльность и тянется к Синдикату
espionage_operation_event = {
	id = ps_covert.2
	title = ps_covert.2.name
	desc = ps_covert.2.desc
	espionage_operation = yes
	picture = GFX_evt_spy_network
	show_sound = event_espionage_concluded
	is_triggered_only = yes

	option = {
		name = ps_covert.2.a
		target = {
			add_modifier = { modifier = ps_shifting_allegiance days = 1800 }
			set_timed_country_flag = { flag = ps_puppet_council days = 3600 }
			add_opinion_modifier = { who = root.owner modifier = opinion_ps_puppet_council }
		}
	}

	after = { destroy_espionage_operation = this }
}

	show_sound = event_espionage_concluded
	is_triggered_only = yes

	option = {

		target = {
			set_timed_country_flag = { flag = ps_recently_decapitated days = 3600 }

}

# ПОДАВИТЬ СМУТУ: помощь союзнику против восстания
espionage_operation_event = {
	id = ps_covert.7
	title = ps_covert.7.name
	desc = ps_covert.7.desc
	espionage_operation = yes
	picture = GFX_evt_spy_network
	show_sound = event_espionage_concluded
	is_triggered_only = yes

	immediate = {
		target = {
			random_situation = {
				limit = { is_situation_type = situation_planetary_revolt }
				save_event_target_as = ps_ally_revolt
			}
		}
	}

	option = {
		name = ps_covert.7.a
		if = {
			limit = { exists = event_target:ps_ally_revolt }
			event_target:ps_ally_revolt = { situation_event = { id = ps_covert.20 } }
		}
		target = {
			add_opinion_modifier = { who = root.owner modifier = opinion_ps_covert_aid }
		}
	}

	after = { destroy_espionage_operation = this }
}
