# abduct pops as slaves
iseeu_operation_abduct_slaves = {
	target = none
	categories = {
		op_cat_subterfuge
		op_cat_economy
	}
	picture = GFX_evt_dark_alley
	desc = iseeu_operation_abduct_slaves_desc
	stages = 2
	resources = {
		category = operations
		cost = {
			energy = 2000
		}
		upkeep = {
			energy = 5
		}
	}
	spy_power_cost = 20
	# spy_network_lvl_requirement = 20
	#remove 20
	potential = {
		owner = {
			has_policy_flag = slavery_allowed
		}
		target = {
			any_owned_pop_group = {
				species = {
					iseeu_is_slave = {
						COUNTRY = root.owner
					}
				}
			}
		}
	}
	allow = {
		custom_tooltip = {
			is_running_espionage_operation = no
			fail_text = operation_one_at_a_time
		}
	}
	abort = {
		always = no
	}
	stage = {
		difficulty = @diff_t0
		icon = GFX_espionage_chapter_icon_target
		event = iseeu_subterfuge.1
	}
	stage = {
		difficulty = @diff_t3
		icon = GFX_espionage_chapter_icon_success
		event = iseeu_subterfuge.5
	}
	on_roll_failed = {
		iseeu_espionage_operation_on_roll_failed = yes
	}
	on_create = {
	}
}
