#####################################################
# Синдикат Сопротивления — КРИЗИС КОНЦА ИГРЫ «КРАСНАЯ БУРЯ»
# Восстание, которое свергает своего хозяина, а затем по одному
# объявляет войну соседям (кроме партизанских империй игроков).
# Формат ситуаций сверен с vanilla 4.x (99_README_SITUATIONS.txt).
#
# situation_ps_red_storm — у государства Бури: 5 уровней силы.
# Уровень растёт со временем, во время войны и (+20) за каждую победу.
#####################################################

situation_ps_red_storm = {
	picture = GFX_ps_evt_partisans
	category = positive
	situation_log_category = urgent_matters
	permanent = yes

	complete_icon = GFX_situation_outcome_revolt
	complete_icon_frame = GFX_situation_outcome_frame_red

	initial_progress = 0

	abort_trigger = {
		OR = {
			NOT = { exists = owner }
			owner = { num_owned_planets < 1 }
			owner = { NOT = { has_country_flag = ps_red_storm } }
		}
	}

	on_abort = {
		hidden_effect = {
			situation_event = { id = ps_storm.50 }
		}
	}

	stages = {
		ps_storm_level_1 = {
			icon = GFX_situation_stage_1
			icon_background = GFX_situation_stage_frame_red
			end = 20
			modifier = {
				leader_lifespan_add = 1000
				army_damage_mult = 0.25
				army_health = 0.25
				country_naval_cap_mult = 0.25
				ship_fire_rate_mult = 0.10
				planet_stability_add = 10
			}
		}
		ps_storm_level_2 = {
			icon = GFX_situation_stage_2
			icon_background = GFX_situation_stage_frame_red
			end = 40
			on_first_enter = { situation_event = { id = ps_storm.3 } }
			modifier = {
				leader_lifespan_add = 1000
				army_damage_mult = 0.40
				army_health = 0.40
				country_naval_cap_mult = 0.50
				ship_fire_rate_mult = 0.20
				ship_hull_mult = 0.10
				planet_stability_add = 10
			}
		}
		ps_storm_level_3 = {
			icon = GFX_situation_stage_3
			icon_background = GFX_situation_stage_frame_red
			end = 60
			on_first_enter = { situation_event = { id = ps_storm.3 } }
			modifier = {
				leader_lifespan_add = 1000
				army_damage_mult = 0.60
				army_health = 0.60
				country_naval_cap_mult = 0.75
				ship_fire_rate_mult = 0.30
				ship_hull_mult = 0.20
				planet_stability_add = 15
			}
		}
		ps_storm_level_4 = {
			icon = GFX_situation_stage_4
			icon_background = GFX_situation_stage_frame_red
			end = 80
			on_first_enter = { situation_event = { id = ps_storm.3 } }
			modifier = {
				leader_lifespan_add = 1000
				army_damage_mult = 0.80
				army_health = 0.80
				country_naval_cap_mult = 1.00
				ship_fire_rate_mult = 0.40
				ship_hull_mult = 0.30
				ship_speed_mult = 0.10
				planet_stability_add = 15
			}
		}
		ps_storm_level_5 = {
			icon = GFX_situation_stage_5
			icon_background = GFX_situation_stage_frame_red
			end = 100
			on_first_enter = { situation_event = { id = ps_storm.3 } }
			modifier = {
				leader_lifespan_add = 1000
				army_damage_mult = 1.00
				army_health = 1.00
				country_naval_cap_mult = 1.50
				ship_fire_rate_mult = 0.50
				ship_hull_mult = 0.50
				ship_speed_mult = 0.20
				planet_stability_add = 20
			}
		}
	}

	monthly_progress = {
		base = 0.2
		modifier = {
			add = 0.3
			desc = ps_storm_progress_war
			owner = { is_at_war = yes }
		}
		modifier = {
			add = 0.3
			desc = ps_storm_progress_size
			owner = { num_owned_planets > 10 }
		}
	}

	approach = {
		name = ps_storm_approach_world_revolution
		icon = GFX_situation_approach_whip
		icon_background = GFX_situation_approach_bg_red
		default = yes
		on_select = {
			custom_tooltip = ps_storm_approach_world_revolution_tt
		}
		ai_weight = {
			base = 1
		}
	}
}
