# Синдикат Сопротивления — скриптовые эффекты

# Изменить уровень проникновения шпионской сети, нацеленной на
# event_target:ps_op_target. Отрицательное AMOUNT — «плата» за операцию.
ps_spend_network = {
	random_spynetwork = {
		limit = {
			owner = { is_same_value = root }
			target = { is_same_value = event_target:ps_op_target }
		}
		add_spy_network_level = $AMOUNT$
	}
}

# Сделать текущую страну вассалом сюзерена (параметр OVERLORD) по соглашению «Режим-клиент».
# ВНИМАНИЕ: синтаксис set_subject_of не удалось сверить с ванильными файлами 4.x
# (нужен файл, где ваниль создаёт вассалов скриптом). Если в error.log будет ошибка
# на этой строке — исправить здесь. Запасной путь: страна получает «Марионеточный
# совет» и +200 к отношению, поэтому примет вассалитет через обычную дипломатию.
ps_make_cell_subject = {
	set_subject_of = {
		who = $OVERLORD$
		agreement_preset = preset_ps_client_regime
	}
}

# КРАСНАЯ БУРЯ: страна (scope) провозглашает Мировую революцию и становится кризисом
ps_storm_ignite = {
	set_country_flag = ps_red_storm
	set_global_flag = ps_storm_happened
	start_situation = { type = situation_ps_red_storm }
	add_resource = { alloys = 5000 energy = 5000 influence = 300 unity = 2000 }
	every_playable_country = {
		limit = { ps_storm_target = yes }
		add_opinion_modifier = { who = prev modifier = opinion_ps_red_storm }
		country_event = { id = ps_storm.2 }
	}
	country_event = { id = ps_storm.1 }
}
