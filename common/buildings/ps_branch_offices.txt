# Синдикат Сопротивления — здания филиалов
# Формат сверен с vanilla common/buildings/14_branch_office_buildings.txt (4.x)
#   triggered_country_modifier  -> действует на ВЛАДЕЛЬЦА ФИЛИАЛА
#   triggered_planet_modifier   -> действует на планету-хозяина
#   job_*_add = 100             -> одна должность (система workforce 4.x)

# Партизанская ячейка: схрон, явка, радиостанция
building_ps_partisan_cell = {
	base_buildtime = @b1_time
	planet_limit = 1

	owner_type = corporate
	category = army

	potential = {
		has_branch_office = yes
		branch_office_owner = { ps_is_partisan_empire = yes }
	}

	# Владельцу филиала: агентура и доход
	triggered_country_modifier = {
		potential = { always = yes }
		spy_network_levels_add = 2
		country_base_energy_produces_add = 2
		mult = value:branch_office_value_for_buildings
	}

	# Планете-хозяину: подпольщики и преступность
	triggered_planet_modifier = {
		potential = { always = yes }
		job_criminal_add = 100
		planet_stability_add = -2
	}

	triggered_planet_modifier = {
		planet_crime_add = 15
		mult = branch_office_owner.value:scripted_modifier_mult|MODIFIER|branch_office_holding_crime_mult|
	}

	resources = {
		category = planet_branch_office_buildings
		cost = {
			minerals = 400
			influence = 15
		}
		upkeep = {
			energy = 2
		}
	}

	ai_weight = {
		factor = 15
	}
}
