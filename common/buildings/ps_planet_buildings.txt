# Синдикат Сопротивления — здания на своих планетах
# Идея из мода Skullduggery: бонусы растут вместе с числом соперников и контактов
# (mult = owner.trigger:num_rivals / num_communications — как в vanilla).

building_ps_resistance_hq = {
	base_buildtime = @b2_time
	category = government
	empire_limit = 1

	building_sets = {
		government
		unity
		urban
	}

	potential = {
		exists = owner
		owner = { ps_is_partisan_empire = yes }
	}

	destroy_trigger = {
		exists = owner
		owner = { NOT = { ps_is_partisan_empire = yes } }
	}

	triggered_country_modifier = {
		potential = { always = yes }
		spy_network_levels_add = 2
	}

	resources = {
		category = planet_buildings
		cost = {
			minerals = 1000
			influence = 50
		}
		upkeep = {
			energy = 5
		}
		# За каждого соперника
		produces = {
			unity = 10
			influence = 0.5
			mult = owner.trigger:num_rivals
		}
	}
}
