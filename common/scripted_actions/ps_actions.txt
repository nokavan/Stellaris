# Синдикат Сопротивления — действия ковчега (как ванильное «Поселиться» / arkship_settle).
# Кнопка в панели флота ковчега и пункт меню по правому клику на свободной планете.

# ОСНОВАТЬ ЯЧЕЙКУ ЗДЕСЬ: свободная пригодная планета в системе ковчега становится вассалом
# по выбранной доктрине (как решение «Основать ячейку»).
ps_found_cell_here = {
	user_scope = fleet
	scope = planet

	on_completed = on_ps_found_cell_here
	tooltip = ps_found_cell_here_button
	icon = GFX_fleet_order_button_drop_off_prisoners
	icon_selected = GFX_fleet_order_drop_off_prisoners

	possible = {
		# This = планета, From = флот ковчега
		custom_tooltip = {
			fail_text = ps_found_cell_here_partisan_tt
			from.controller = { ps_is_partisan_empire = yes }
		}
		custom_tooltip = {
			fail_text = ps_found_cell_here_free_tt
			ps_free_habitable_planet = { OWNER = from.controller }
		}
		custom_tooltip = {
			fail_text = ps_found_cell_here_space_tt
			solar_system = {
				OR = {
					NOT = { exists = space_owner }
					space_owner = { is_same_value = from.controller }
				}
			}
		}
		custom_tooltip = {
			fail_text = decision_ps_planet_cell_cooldown
			from.controller = { NOT = { has_country_flag = ps_planet_cell_cd } }
		}
	}

	button_visible = {
		is_arkship_fleet = yes
		controller = { ps_is_partisan_empire = yes }
	}

	button_clickable = {
		custom_tooltip = {
			fail_text = decision_ps_planet_cell_cooldown
			controller = { NOT = { has_country_flag = ps_planet_cell_cd } }
		}
	}

	hide_from_context_menu = {
		NOT = { from.controller = { ps_is_partisan_empire = yes } }
	}

	activity_key = ps_found_cell_here_activity_key
	context_menu_name = ps_found_cell_here
	required_progress = 30
	progress_activity_key = ps_found_cell_here_progress_activity_key
	required_progress_key = ps_found_cell_here_required_progress_key
	category = arkship_settle
	category_limit = 1

	cost = {
		alloys = 500
		food = 300
		influence = 100
	}

	slot = 5
}
