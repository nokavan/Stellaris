# Скоупы сверены с vanilla common/on_actions/00_on_actions.txt (4.x)

on_game_start_country = {
	events = {
		ps.1	# Пролог: Архив Сопротивления
		ps.5	# Пролог: Ковчег Сопротивления
	}
}

# this = country
on_yearly_pulse_country = {
	events = {
		ps.14	# Сюзерен: Присяга Синдиката, каперские рейды, слоты владений
		ps.17	# Снять слоты владений у бывших вассалов
		ps.50	# Диспетчер сюжета и агентурных отчётов
		ps_revolt.30	# Фонд освобождения: поддержка чужих восстаний
	}
}

# root = planet, from = owner, fromfrom = occupier
on_planet_occupied = {
	events = {
		ps.30	# Партизаны поднимаются
	}
}

# root = planet, from = owner, fromfrom = former occupier
on_planet_returned = {
	events = {
		ps.32	# Планета освобождена
	}
}

# this = planet, from = branch office owner
on_branch_office_established = {
	events = {
		ps.60	# +20 слотов зданий филиала
	}
}

# this = planet, from = branch office owner
on_branch_office_closed = {
	events = {
		ps.61	# убрать слоты
	}
}

# this = country
on_monthly_pulse_country = {
	events = {
		ps.70	# Ковчег: стоянка Сопротивления
	}
}

	}

	}
}
