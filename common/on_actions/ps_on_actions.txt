# Синдикат Сопротивления — on_actions (добавляются к ванильным, не заменяют их)
# Скоупы сверены с vanilla common/on_actions/00_on_actions.txt (4.x)

on_game_start_country = {
	events = {
		ps.1	# Пролог: Архив Сопротивления
		ps.5	# Пролог: Ковчег Сопротивления
		ps_megaark.1	# Мега-ковчеги (встроенный Mega Arkships)
	}
}

# this = country
on_yearly_pulse_country = {
	events = {
		ps.14	# Сюзерен: Присяга Синдиката, каперские рейды, слоты владений
		ps.17	# Снять слоты владений у бывших вассалов
		ps.50	# Диспетчер сюжета и агентурных отчётов
		ps_revolt.30	# Фонд освобождения: поддержка чужих восстаний
		ps_storm.30	# Красная Буря: следующая война с соседом
		ps_storm.40	# Красная Буря может родиться в империи ИИ
		ps_vm.90	# Вассальные проекты: модификаторы вассалов
	}
}

# root = planet, from = owner, fromfrom = occupier
on_planet_occupied = {
	events = {
		ps.30	# Партизаны поднимаются
	}
}

# root = planet, from = owner, fromfrom = former occupier
on_planet_returned = {
	events = {
		ps.32	# Планета освобождена
	}
}

# this = planet, from = branch office owner
on_branch_office_established = {
	events = {
		ps.60	# +20 слотов зданий филиала
	}
}

# this = planet, from = branch office owner
on_branch_office_closed = {
	events = {
		ps.61	# убрать слоты
	}
}

# this = country
on_monthly_pulse_country = {
	events = {
		ps.70	# Ковчег: стоянка Сопротивления
		ps_mega.20	# Теневая логистика: автомаскировка
		ps_megaark.1	# Мега-ковчеги: буст новых ковчегов
		ps_so.1	# Теневой сюзерен: +5% единства за вассала
	}
}

# this = rebel country, from = colony, fromfrom = war
on_rebels_take_planet = {
	events = {
		ps_revolt.40	# Восставшие победили: бафф и награда покровителям
	}
}

# this = subject, from = overlord
on_becoming_subject = {
	events = {
		ps.16	# 20 слотов владений у вассала партизанского сюзерена
		ps_vm.91	# Вассальные проекты сюзерена — новому вассалу
	}
}

# root = country, from = megastructure, fromfrom = system
on_megastructure_built = {
	events = {
		ps_mega.5	# Мегаструктурное заселение
	}
}

# root = country, from = megastructure, fromfrom = system
on_megastructure_upgraded = {
	events = {
		ps_mega.5	# Мегаструктурное заселение (кольцевые миры, аркологии и т. п.)
	}
}
