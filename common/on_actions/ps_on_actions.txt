		ps.50	# Диспетчер сюжета и агентурных отчётов
	}
}

# this = subject, from = overlord

	events = {
		ps.13	# Сразу предложить роль новому вассалу
	}
}

	}
}

# root = planet, from = owner, fromfrom = former occupier

	}
