# Синдикат Сопротивления — общие триггеры

# Одно из партизанских происхождений (сюжетная цепочка)
ps_has_partisan_origin = {
	OR = {
		has_origin = origin_ps_partisan_legacy
		ps_is_partisan_ark = yes
	}
}

# Партизанский ковчег: наше происхождение «Ковчег Сопротивления» ИЛИ
# «Мировой ковчег» из мода XHK World Ark у партизанской корпорации
# (флаг ставит патч совместимости compat/xhk_worldark_patch)
ps_is_partisan_ark = {
	OR = {
		has_origin = origin_ps_partisan_ark
		has_country_flag = ps_partisan_worldark
	}
}

# Империя, ведущая партизанскую сеть (происхождение или цивик)
ps_is_partisan_empire = {
	OR = {
		has_origin = origin_ps_partisan_legacy
		has_origin = origin_ps_partisan_ark
		has_country_flag = ps_partisan_worldark
		has_valid_civic = civic_ps_partisan_network
	}
}

# Сюзерен, умеющий специализировать вассалов
ps_is_shadow_overlord = {
	OR = {
		has_origin = origin_ps_partisan_legacy
		has_origin = origin_ps_partisan_ark
		has_country_flag = ps_partisan_worldark
		has_valid_civic = civic_ps_shadow_overlord
	}
}

# У сюзерена есть вассал «Шпионская резидентура» (флаг ставит перк)
ps_has_spy_ring_subject = {
	any_playable_country = {
		is_subject = yes
		overlord = { is_same_value = root }
		has_country_flag = ps_residency_active
	}
}

# ---- Здания филиалов (scope = планета-хозяин) ----

# Хозяин планеты — союзник владельца филиала
ps_branch_host_is_friend = {
	exists = owner
	exists = branch_office_owner
	branch_office_owner = {
		OR = {
			is_overlord_to = prev.owner
			is_in_federation_with = prev.owner
			has_defensive_pact = prev.owner
			is_friendly_to = prev.owner
		}
	}
}

# Хозяин планеты — соперник/враг владельца филиала
ps_branch_host_is_enemy = {
	exists = owner
	exists = branch_office_owner
	branch_office_owner = {
		OR = {
			is_rival = prev.owner
			is_at_war_with = prev.owner
			opinion_level = {
				who = prev.owner
				level <= poor
			}
		}
	}
}

# ---- Панель ячеек ----
# У страны (сюзерена) есть выбранная в панели ячейка-вассал.
# Вызывать в скоупе сюзерена; prevprev = сюзерен (scope: сюзерен -> вассал -> overlord).
ps_has_selected_cell = {
	any_playable_country = {
		is_subject = yes
		has_country_flag = ps_cell_selected
		overlord = { is_same_value = prevprev }
	}
}

# ---- Ковчег: высадка ячейки ----
# Есть стоящий лагерем ковчег, а в его системе — свободная пригодная планета,
# и система ничья или наша. Scope = страна.
ps_ark_has_seedable_world = {
	any_owned_fleet = {
		is_arkship_fleet = yes
		exists = solar_system
		any_owned_ship = {
			is_arkship_ship = yes
			is_system_locked = yes
		}
		solar_system = {
			OR = {
				NOT = { exists = space_owner }
				space_owner = { is_same_value = prevprev.owner }
			}
			any_system_planet = {
				is_colonizable = yes
				NOT = { exists = owner }
			}
		}
	}
}

# ---- Кризисы ----
# У страны уже идёт кризис Синдиката (scope = страна)
ps_has_active_crisis = {
	any_situation = {
		OR = {
			is_situation_type = situation_ps_general_strike
			is_situation_type = situation_ps_insurgency
			is_situation_type = situation_ps_financial_collapse
			is_situation_type = situation_ps_legitimacy
			is_situation_type = situation_ps_naval_mutiny
		}
	}
}

# ---- Красная Буря ----
# Партизанская империя под управлением игрока (Буря её не трогает)
ps_is_partisan_player = {
	is_ai = no
	ps_is_partisan_empire = yes
}

# Страна — допустимая цель Бури: обычная империя, не сама Буря и не её вассал,
# не партизанская империя игрока и не вассал такой империи (scope = страна)
ps_storm_target = {
	is_country_type = default
	NOT = { has_country_flag = ps_red_storm }
	NOT = { ps_is_partisan_player = yes }
	NOT = {
		AND = {
			is_subject = yes
			overlord = {
				OR = {
					has_country_flag = ps_red_storm
					ps_is_partisan_player = yes
				}
			}
		}
	}
}
