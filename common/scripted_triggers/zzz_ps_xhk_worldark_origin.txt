# Синдикат Сопротивления: XHK World Ark узнаёт наше 3-е происхождение и создаёт для него Мировой ковчег.
# Переопределяет триггер XHK по ключу («Синдикат» должен стоять в плейсете НИЖЕ XHK World Ark).
# Без XHK триггер не используется.
#
# Страховка: наше происхождение считается «Мировым ковчегом» только первые 20 дней (пока XHK
# создаёт ковчег, флаг ps_wark_pending — событие ps_wark.1) или если технология Мирового ковчега
# действительно получена. Если создание не удалось (например, неверный порядок модов), XHK не
# «поглотит» обычный ковчег, и игра продолжится на нём с бонусами «Ковчега Сопротивления».
xhk_has_worldark_origin = {
	OR = {
		has_origin = origin_xhk_worldark
		AND = {
			has_origin = origin_ps_partisan_worldark
			OR = {
				NOT = { has_country_flag = ps_wark_started }	# старт игры, до события ps_wark.1
				has_country_flag = ps_wark_pending				# первые 20 дней
				has_technology = xhk_tech_worldark_1			# Мировой ковчег создан
			}
		}
	}
}
