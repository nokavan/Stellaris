# Синдикат Сопротивления — панель управления ячейками (вассалами)
# Формат решений — как в модах Feudal Expanded (owned_planets_only/potential/allow/effect).
# root = планета.

# Открыть панель ячеек (на столице)
decision_ps_cell_panel = {
	owned_planets_only = yes

	potential = {
		is_capital = yes
		owner = { ps_is_shadow_overlord = yes }
	}

	allow = {
		custom_tooltip = {
			fail_text = decision_ps_no_cells
			owner = {
				any_playable_country = {
					is_subject = yes
					overlord = { is_same_value = root.owner }
				}
			}
		}
	}

	effect = {
		owner = { country_event = { id = ps_panel.1 } }
	}
}

# Передать эту планету выбранной в панели ячейке
decision_ps_transfer_planet = {
	owned_planets_only = yes

	resources = {
		category = decisions
		cost = {
			influence = 25
		}
	}

	potential = {
		is_capital = no
		owner = { ps_is_shadow_overlord = yes }
	}

	allow = {
		is_colony = yes
		custom_tooltip = {
			fail_text = decision_ps_no_selected_cell
			owner = { ps_has_selected_cell = yes }
		}
		custom_tooltip = {
			fail_text = decision_ps_planet_occupied
			exists = controller
			controller = { is_same_value = root.owner }
		}
	}

	effect = {
		custom_tooltip = decision_ps_transfer_planet_tooltip
		hidden_effect = {
			carrier_event = { id = ps_panel.50 }
		}
	}
}

# КОВЧЕГ: высадить ячейку — основать новое государство-вассала на свободной
# планете в системе, где стоит лагерем наш ковчег.
decision_ps_ark_seed_cell = {
	owned_planets_only = yes

	resources = {
		category = decisions
		cost = {
			alloys = 1000
			food = 500
			influence = 150
		}
	}

	potential = {
		always = no	# заменено общим решением «Основать ячейку» (выбор планеты: ковчег, стоянки, границы)
		is_capital = yes
		owner = { ps_is_partisan_ark = yes }
	}

	allow = {
		custom_tooltip = {
			fail_text = decision_ps_ark_seed_cooldown
			owner = { NOT = { has_country_flag = ps_ark_seed_cooldown } }
		}
		custom_tooltip = {
			fail_text = decision_ps_ark_no_camp_world
			owner = { ps_ark_has_seedable_world = yes }
		}
	}

	effect = {
		custom_tooltip = decision_ps_ark_seed_cell_tooltip
		hidden_effect = {
			owner = { set_timed_country_flag = { flag = ps_ark_seed_cooldown days = 1800 } }
			carrier_event = { id = ps_ark.10 }
		}
	}
}

# ТЕНЕВАЯ ЛОГИСТИКА: включить автомаскировку строителей и кораблей снабжения
decision_ps_shadow_logistics_on = {
	owned_planets_only = yes

	resources = {
		category = decisions
		cost = {
			influence = 50
		}
	}

	potential = {
		is_capital = yes
		owner = {
			ps_is_partisan_empire = yes
			NOT = { has_country_flag = ps_shadow_logistics }
		}
	}

	effect = {
		owner = {
			set_country_flag = ps_shadow_logistics
			add_modifier = { modifier = ps_shadow_logistics }
		}
		custom_tooltip = decision_ps_shadow_logistics_on_tooltip
	}
}

decision_ps_shadow_logistics_off = {
	owned_planets_only = yes

	potential = {
		is_capital = yes
		owner = {
			ps_is_partisan_empire = yes
			has_country_flag = ps_shadow_logistics
		}
	}

	effect = {
		owner = {
			remove_country_flag = ps_shadow_logistics
			remove_modifier = ps_shadow_logistics
			every_owned_fleet = {
				limit = {
					ps_is_logistics_fleet = yes
					is_cloaked = yes
				}
				set_cloaking_active = no
			}
		}
		custom_tooltip = decision_ps_shadow_logistics_off_tooltip
	}
}

# КОВЧЕГ: включить / выключить стелс-модуль
decision_ps_ark_cloak_off = {
	owned_planets_only = yes

	potential = {
		is_capital = yes
		owner = {
			NOT = { has_country_flag = ps_ark_cloak_off }
			any_owned_starbase = {
				is_arkship_starbase = yes
				has_starbase_module = arkship_ps_stealth_field
			}
		}
	}

	effect = {
		custom_tooltip = decision_ps_ark_cloak_off_tooltip
		hidden_effect = {
			owner = {
				set_country_flag = ps_ark_cloak_off
				every_owned_starbase = {
					limit = {
						is_arkship_starbase = yes
						has_starbase_module = arkship_ps_stealth_field
					}
					fleet = { set_cloaking_active = no }
				}
			}
		}
	}
}

decision_ps_ark_cloak_on = {
	owned_planets_only = yes

	potential = {
		is_capital = yes
		owner = {
			has_country_flag = ps_ark_cloak_off
			any_owned_starbase = {
				is_arkship_starbase = yes
				has_starbase_module = arkship_ps_stealth_field
			}
		}
	}

	effect = {
		custom_tooltip = decision_ps_ark_cloak_on_tooltip
		hidden_effect = {
			owner = {
				remove_country_flag = ps_ark_cloak_off
				every_owned_starbase = {
					limit = {
						is_arkship_starbase = yes
						has_starbase_module = arkship_ps_stealth_field
					}
					fleet = {
						if = {
							limit = { has_cloaking_strength > 0 }
							set_cloaking_active = yes
						}
					}
				}
			}
		}
	}
}

# ВЫДЕЛИТЬ ПЛАНЕТУ В ЯЧЕЙКУ: наша колония (не столица) становится новым вассалом
decision_ps_release_planet_cell = {
	owned_planets_only = yes

	resources = {
		category = decisions
		cost = {
			influence = 75
		}
	}

	potential = {
		is_capital = no
		owner = { ps_is_partisan_empire = yes }
	}

	allow = {
		is_colony = yes
		custom_tooltip = {
			fail_text = decision_ps_planet_occupied
			exists = controller
			controller = { is_same_value = root.owner }
		}
		custom_tooltip = {
			fail_text = decision_ps_planet_cell_cooldown
			owner = { NOT = { has_country_flag = ps_planet_cell_cd } }
		}
	}

	effect = {
		custom_tooltip = decision_ps_release_planet_cell_tooltip
		hidden_effect = {
			owner = {
				save_event_target_as = ps_seed_owner
				set_timed_country_flag = { flag = ps_planet_cell_cd days = 1080 }
			}
			ps_cell_from_own_colony = yes
			event_target:ps_seed_owner = { country_event = { id = ps_mega.8 } }
		}
	}
}
