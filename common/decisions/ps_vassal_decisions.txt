decision_ps_vp_chancellery_1 = {
	owned_planets_only = yes

	resources = {
		category = decisions
		cost = {
			energy = 2000
			alloys = 500
			influence = 100
		}
	}

	potential = {
		is_capital = yes
		owner = {
			NOT = { has_country_flag = ps_vp_chancellery_1 }
		}
	}

	allow = {
		custom_tooltip = {
			fail_text = decision_ps_no_cells
			owner = { any_subject = { always = yes } }
		}
		custom_tooltip = {
			fail_text = ps_vp_cooldown
			owner = { NOT = { has_country_flag = ps_vp_cd } }
		}
	}

	effect = {
		custom_tooltip = decision_ps_vp_chancellery_1_tooltip
		hidden_effect = {
			owner = {
				set_country_flag = ps_vp_chancellery_1
				set_timed_country_flag = { flag = ps_vp_cd days = 720 }
				add_modifier = { modifier = ps_vp_chancellery_ov_1 days = -1 }
				country_event = { id = ps_vm.90 }
			}
		}
	}

	ai_weight = {
		weight = 0
	}
}
